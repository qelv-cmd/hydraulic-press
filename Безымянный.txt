            function initAudio() {
                if (audioCtx) return;
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                if (!AudioCtx) return;

                audioCtx = new AudioCtx();
                pumpOsc = audioCtx.createOscillator();
                valveOsc = audioCtx.createOscillator();
                pumpFilter = audioCtx.createBiquadFilter();
                pumpGain = audioCtx.createGain();

                pumpOsc.type = 'sawtooth';
                pumpOsc.frequency.value = 54;

                valveOsc.type = 'triangle';
                valveOsc.frequency.value = 320;

                pumpFilter.type = 'lowpass';
                pumpFilter.frequency.value = 180;
                pumpFilter.Q.value = 5.0;

                pumpGain.gain.value = 0;

                pumpOsc.connect(pumpFilter);
                valveOsc.connect(pumpFilter);
                pumpFilter.connect(pumpGain);
                pumpGain.connect(audioCtx.destination);

                pumpOsc.start();
                valveOsc.start();
            }

            function triggerDestructionSound(behavior) {
                if (!state.soundEnabled || !audioCtx) return;
                try {
                    const now = audioCtx.currentTime;
                    const bufferSize = audioCtx.sampleRate * 0.55;
                    const noiseBuffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
                    const output = noiseBuffer.getChannelData(0);
                    for (let i = 0; i < bufferSize; i++) {
                        output[i] = (Math.random() * 2 - 1) * Math.exp(-i / (audioCtx.sampleRate * 0.14));
                    }
                    const whiteNoise = audioCtx.createBufferSource();
                    whiteNoise.buffer = noiseBuffer;

                    const filter = audioCtx.createBiquadFilter();
                    filter.type = (behavior === 'brittle' || behavior === 'electric') ? 'highpass' : 'lowpass';
                    filter.frequency.value = (behavior === 'brittle' || behavior === 'electric') ? 1900 : 460;

                    const nGain = audioCtx.createGain();
                    nGain.gain.setValueAtTime(0.38, now);
                    nGain.gain.exponentialRampToValueAtTime(0.001, now + 0.54);

                    whiteNoise.connect(filter);
                    filter.connect(nGain);
                    nGain.connect(audioCtx.destination);
                    whiteNoise.start(now);

                    // Sub-bass shockwave boom
                    const sub = audioCtx.createOscillator();
                    const subGain = audioCtx.createGain();
                    sub.type = 'sine';
                    sub.frequency.setValueAtTime(125, now);
                    sub.frequency.exponentialRampToValueAtTime(24, now + 0.45);
                    subGain.gain.setValueAtTime(0.36, now);
                    subGain.gain.exponentialRampToValueAtTime(0.001, now + 0.46);
                    sub.connect(subGain);
                    subGain.connect(audioCtx.destination);
                    sub.start(now);
                    sub.stop(now + 0.47);
                } catch (e) {}
            }

            function triggerLaserPulseSound() {
                if (!state.soundEnabled || !audioCtx) return;
                try {
                    const now = audioCtx.currentTime;
                    const osc = audioCtx.createOscillator();
                    const g = audioCtx.createGain();
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(760, now);
                    osc.frequency.exponentialRampToValueAtTime(95, now + 0.16);
                    g.gain.setValueAtTime(0.18, now);
                    g.gain.exponentialRampToValueAtTime(0.001, now + 0.17);
                    osc.connect(g);
                    g.connect(audioCtx.destination);
                    osc.start(now);
                    osc.stop(now + 0.18);
                } catch (e) {}
            }

            function updateHydraulicAudio(isMoving, loadNorm, timeScale) {
                if (!state.soundEnabled || !audioCtx || !pumpGain) return;
                const now = audioCtx.currentTime;
                if (isMoving) {
                    pumpGain.gain.setTargetAtTime(0.06 + loadNorm * 0.06, now, 0.04);
                    pumpOsc.frequency.setTargetAtTime((48 + loadNorm * 85 + state.pumpSpeedPct * 0.25) * (0.4 + 0.6 * timeScale), now, 0.05);
                    valveOsc.frequency.setTargetAtTime((280 + loadNorm * 720) * (0.4 + 0.6 * timeScale), now, 0.05);
                    pumpFilter.frequency.setTargetAtTime(160 + loadNorm * 680, now, 0.05);
                } else {
                    pumpGain.gain.setTargetAtTime(0, now, 0.04);
                }
            }

            // --- Update Deformed Specimen Vertices (Affected by Die Geometry & Dent!) ---
            function updateSpecimenGeometry(spec, strain, loadNorm) {
                const currentHeight = Math.max(0.08, spec.restHeight * (1.0 - strain));
                const centerY = ANVIL_TOP_Y + currentHeight * 0.5;
                const volumeBulge = 1.0 / Math.sqrt(Math.max(0.18, 1.0 - strain * 0.85));

                const positions = specGeo.attributes.position;
                const colors = specGeo.attributes.color;
                const isThermal = state.visMode === 'thermal';

                for (let i = 0; i < specVertCount; i++) {
                    const ny = baseNormY[i]; // -0.5 to +0.5
                    const angle = baseAngle[i];
                    const radRatio = baseRadRatio[i];

                    let y = centerY + ny * currentHeight;
                    let r = 0;

                    if (radRatio > 0.5) {
                        const endPinch = Math.cos(ny * Math.PI);
                        let localRadius = spec.restRadius * (1.0 + (volumeBulge - 1.0) * (0.24 + 0.76 * endPinch));

                        if (spec.bucklingWaves > 0 && strain > 0.04) {
                            const wave = Math.sin((ny + 0.5) * Math.PI * spec.bucklingWaves * 2.0);
                            localRadius += wave * spec.restRadius * strain * 0.34 * endPinch;
                        }

                        // Local laser/hammer tap dent around equator
                        if (state.dentDepth > 0.001) {
                            const eqFactor = Math.exp(-Math.pow(ny * 4.5, 2));
                            const angFactor = Math.max(0, Math.cos(angle - 0.5));
                            localRadius -= state.dentDepth * eqFactor * Math.pow(angFactor, 3);
                        }

                        // Die-specific top deformation!
                        if (ny > 0.15 && strain > 0.02) {
                            const topWeight = (ny - 0.15) / 0.35;
                            if (state.dieType === 'wedge') {
                                // V-wedge splits along X axis (z ~ 0)
                                const zNorm = Math.abs(Math.sin(angle));
                                y -= (1.0 - zNorm) * strain * 0.35 * topWeight;
                                localRadius += zNorm * strain * 0.18 * topWeight;
                            } else if (state.dieType === 'spikes') {
                                // Spike matrix ripples the top crown
                                const spikeRipple = Math.abs(Math.sin(angle * 4.0));
                                y -= spikeRipple * strain * 0.26 * topWeight;
                            }
                        }

                        if (loadNorm > 0.78 && (spec.behavior === 'brittle' || spec.behavior === 'explosive')) {
                            localRadius += (Math.random() - 0.5) * 0.016 * loadNorm;
                        }

                        r = Math.max(0.05, localRadius);
                    } else if (ny > 0 && strain > 0.02) {
                        // Top cap center indented by Wedge or Spikes
                        if (state.dieType === 'wedge' || state.dieType === 'spikes') {
                            y -= strain * 0.28;
                        }
                    }

                    positions.setXYZ(i, Math.cos(angle) * r, y, Math.sin(angle) * r);

                    // Von Mises Thermal Heatmap
                    if (isThermal) {
                        let concentration = 0.72 + 0.28 * Math.abs(ny * 2);
                        if (state.dieType === 'wedge') {
                            concentration *= (1.35 - 0.45 * Math.abs(Math.sin(angle)));
                        } else if (state.dieType === 'spikes') {
                            concentration *= (1.25 - 0.35 * Math.abs(Math.sin(angle * 4)));
                        }
                        const s = Math.min(1.0, loadNorm * concentration);
                        let cr, cg, cb;
                        if (s < 0.33) {
                            const t = s / 0.33;
                            cr = 0.05; cg = 0.35 + 0.6 * t; cb = 0.98 - 0.45 * t;
                        } else if (s < 0.66) {
                            const t = (s - 0.33) / 0.33;
                            cr = 0.05 + 0.93 * t; cg = 0.95; cb = 0.53 - 0.48 * t;
                        } else {
                            const t = (s - 0.66) / 0.34;
                            cr = 1.0; cg = 0.95 - 0.85 * t; cb = 0.05;
                        }
                        colors.setXYZ(i, cr, cg, cb);
                    }
                }

                positions.needsUpdate = true;
                if (isThermal) colors.needsUpdate = true;
                specGeo.computeVertexNormals();

                innerCoreMesh.position.y = centerY;
                innerCoreMesh.scale.set(1 + strain * 0.6, currentHeight, 1 + strain * 0.6);

                if (gyroGroup.visible) {
                    gyroGroup.position.y = centerY;
                    const gScale = 1.0 + strain * 0.45;
                    gyroGroup.scale.set(gScale, 1.0 - strain * 0.55, gScale);
                    gyroRing1.rotation.y += 0.04 * (1 + loadNorm * 4);
                    gyroRing1.rotation.z += 0.025 * (1 + loadNorm * 4);
                    gyroRing2.rotation.y -= 0.05 * (1 + loadNorm * 4);
                }
            }

            function completeCurrentContract() {
                const earned = Math.round(500 * state.combo);
                state.score += earned;
                state.combo = Math.min(3.0, +(state.combo + 0.25).toFixed(2));
                scoreBadge.textContent = state.score;
                contractComboEl.textContent = `x${state.combo.toFixed(1)}`;
                showToast('КОНТРАКТ ВЫПОЛНЕН!', `+${earned} ОЧКОВ • СЛЕДУЮЩЕЕ ИСПЫТАНИЕ`, true);

            function onContractSpecimenDestroyed() {
                const c = CONTRACTS[state.contractIdx];
                if (c.id === 1) {
                    failCurrentContract('НЕОН-ГЕЛЬ ЛОПНУЛ! НУЖЕН ЮВЕЛИРНЫЙ КОНТРОЛЬ');
                } else if (c.id === 2) {
                    if (state.contractPeakForce <= 320) {
                        completeCurrentContract();
                    } else {
                        failCurrentContract(`ПИК УСИЛИЯ ${Math.round(state.contractPeakForce)} т > 320 т!`);
                    }
                } else if (c.id === 3) {
                    if (state.usedSlamThisAttempt) {
                        completeCurrentContract();
                    } else {
                        failCurrentContract('НУЖНО БЫЛО ИСПОЛЬЗОВАТЬ УДАР «💥 1000T»!');
                    }
                } else if (c.id === 4) {
                    failCurrentContract('ПЛАЗМА-ЯДРО ВЗОРВАЛОСЬ! СБРАСЫВАЙ РАНЬШЕ');
                } else if (c.id === 5) {
                    state.marathonCrushes++;
                    contractProgressEl.style.width = `${(state.marathonCrushes / 3) * 100}%`;
                    if (state.marathonCrushes >= 3) {
                        completeCurrentContract();
                    } else {
                        setTimeout(() => {
                            if (state.gameMode === 'contracts' && CONTRACTS[state.contractIdx].id === 5) {
                                const nextSpec = (state.specIndex + 1) % 6;
                                spawnSpecimen(nextSpec);
                            }
                        }, 450);
                    }
                }
            }

            function stepContracts(dt) {
                if (state.gameMode !== 'contracts') return;
                const c = CONTRACTS[state.contractIdx];

                state.contractTimer -= dt;
                contractTimerEl.textContent = `${Math.max(0, state.contractTimer).toFixed(1)}с`;
                if (state.contractTimer <= 0) {
                    failCurrentContract('ВРЕМЯ ИСТЕКЛО!');
                    return;
                }

                const deformPct = state.totalStrain * 100;
                if (state.forceTons > state.contractPeakForce) {
                    state.contractPeakForce = state.forceTons;
                }

                if (c.id === 1 && !state.destroyed) {
                    // Hold 62% ± 3% for 2.5s
                    if (deformPct >= 59 && deformPct <= 65) {
                        state.contractHoldTime += dt;
                    } else {
                        state.contractHoldTime = Math.max(0, state.contractHoldTime - dt * 0.8);
                    }
                    const pct = Math.min(100, (state.contractHoldTime / 2.5) * 100);
                    contractProgressEl.style.width = `${pct}%`;
                    if (state.contractHoldTime >= 2.5) {
                        completeCurrentContract();
                    }
                } else if (c.id === 2 && !state.destroyed) {
                    const pct = Math.min(100, (deformPct / 38) * 100);
                    contractProgressEl.style.width = `${pct}%`;
                } else if (c.id === 4 && !state.destroyed) {
                    if (deformPct >= 82 && deformPct < 90) {
                        state.contractArmed = true;
                        contractProgressEl.style.width = '75%';
                        hintText.textContent = 'КРИТИЧЕСКИЙ РЕЗОНАНС! ЖМИ «⬆️ СБРОС» ПРЯМО СЕЙЧАС!';
                    } else if (state.contractArmed && deformPct < 45) {
                        contractProgressEl.style.width = '100%';
                        completeCurrentContract();
                    }
                }
            }

            // --- Main Physics Step ---
            function stepPhysics(rawDt) {
                // Slow-Mo time scaling
                if (state.autoSlowMoTimer > 0) {
                    state.autoSlowMoTimer = Math.max(0, state.autoSlowMoTimer - rawDt);
                }
                const isSlowMoActive = state.manualSlowMo || state.autoSlowMoTimer > 0;
                const targetScale = isSlowMoActive ? 0.25 : 1.0;
                state.timeScale += (targetScale - state.timeScale) * 0.22;

                slowmoBadge.classList.toggle('hidden', !isSlowMoActive);
                viewportWrap.classList.toggle('slowmo-vignette', isSlowMoActive);

                const dt = rawDt * state.timeScale;
                const spec = SPECIMENS[state.specIndex];
                const speedFactor = state.pumpSpeedPct / 100;

                // Platen Kinematics
                const specimenTopRestY = ANVIL_TOP_Y + spec.restHeight * (1.0 - state.plasticStrain);
                const inContact = !state.destroyed && (state.platenY <= specimenTopRestY + 0.005);

                if (state.dragMode !== 'platen') {
                    if (state.slamActive) {
                        state.platenVy = -6.2;
                        state.platenY += state.platenVy * dt;
                        if (state.platenY <= MIN_PLATEN_Y || state.destroyed) {
                            state.slamActive = false;
                        }
                    } else if (state.isPressing) {
                        const resistanceSlowdown = inContact ? Math.max(0.25, 1.0 - (state.forceTons / 1250)) : 1.0;
                        const targetVy = -0.95 * speedFactor * resistanceSlowdown;
                        if (state.forceTons < state.maxLimitTons || !inContact) {
                            state.platenVy += (targetVy - state.platenVy) * 0.26;
                            state.platenY += state.platenVy * dt;
                        } else {
                            state.platenVy = 0;
                        }
                    } else if (state.isReleasing) {
                        const targetVy = 1.45 * Math.max(0.6, speedFactor);
                        state.platenVy += (targetVy - state.platenVy) * 0.26;
                        state.platenY += state.platenVy * dt;
                        if (state.platenY >= MAX_PLATEN_Y) {
                            state.platenY = MAX_PLATEN_Y;
                            state.platenVy = 0;
                            state.isReleasing = false;
                        }
                    } else {
                        if (inContact && spec.behavior === 'hyperelastic' && state.elasticStrain > 0.02) {
                            state.platenVy = state.elasticStrain * 2.4;
                            state.platenY += state.platenVy * dt;
                        } else {
                            state.platenVy *= 0.76;
                        }
                    }
                }

                const topLimit = Math.max(MAX_PLATEN_Y, ANVIL_TOP_Y + spec.restHeight + 0.36);
                state.platenY = Math.max(MIN_PLATEN_Y, Math.min(topLimit, state.platenY));
                movingPlatenGroup.position.y = state.platenY;

                // Specimen Deformation & Tonnage Solver
                if (!state.destroyed) {
                    const currentGap = Math.max(0.08, state.platenY - ANVIL_TOP_Y);
                    const compressedHeight = Math.min(spec.restHeight, currentGap);
                    const rawStrain = Math.max(0, (spec.restHeight - compressedHeight) / spec.restHeight);

                    if (spec.behavior === 'ductile' || spec.behavior === 'superalloy') {
                        if (rawStrain > state.plasticStrain) state.plasticStrain = rawStrain;
                        state.elasticStrain = Math.max(0, rawStrain - state.plasticStrain * 0.95);
                        state.totalStrain = state.plasticStrain;
                    } else if (spec.behavior === 'crumble' || spec.behavior === 'composite') {
                        if (rawStrain > state.plasticStrain) state.plasticStrain = rawStrain * 0.68;
                        state.elasticStrain = Math.max(0, rawStrain - state.plasticStrain);
                        state.totalStrain = Math.max(rawStrain, state.plasticStrain);
                    } else {
                        state.plasticStrain = 0;
                        state.elasticStrain = rawStrain;
                        state.totalStrain = rawStrain;
                    }

                    // Die concentration factor (Wedge & Spikes concentrate stress!)
                    const dieMultiplier = state.dieType === 'flat' ? 1.0 : (state.dieType === 'wedge' ? 1.38 : 1.25);

                    if (inContact && rawStrain > 0.002) {
                        const activeStrain = (spec.behavior === 'ductile' || spec.behavior === 'superalloy')
                            ? Math.max(0.02, rawStrain - state.plasticStrain * 0.85 + state.plasticStrain * 0.38)
                            : rawStrain;

                        let nonLinearFactor = 1.0 + Math.pow(rawStrain, 2.1) * 2.3;
                        // Sandwich composite gives 3 distinct load spikes!
                        if (spec.behavior === 'composite') {
                            nonLinearFactor += Math.sin(rawStrain * Math.PI * 5) * 0.25;
                        }
                        const computedForce = activeStrain * spec.stiffnessTons * nonLinearFactor * (state.dieType === 'flat' ? 1.0 : 0.82);
                        state.forceTons = Math.min(1000, computedForce);
                    } else {
                        state.forceTons *= 0.74;
                        if (state.forceTons < 0.1) state.forceTons = 0;
                    }

                    const currentRadius = spec.restRadius / Math.sqrt(Math.max(0.2, 1.0 - state.totalStrain * 0.85));
                    const contactAreaFactor = state.dieType === 'flat' ? 1.0 : (state.dieType === 'wedge' ? 0.25 : 0.18);
                    const areaM2 = Math.PI * Math.pow(currentRadius * 0.15, 2) * contactAreaFactor;
                    state.pressureMpa = (state.forceTons * 9.81 * 1000) / Math.max(0.0005, areaM2 * 1e6);

                    // Thermodynamic Heating from plastic work & friction
                    if (inContact && Math.abs(state.platenVy) > 0.03) {
                        state.temperatureC = Math.min(980, state.temperatureC + state.forceTons * Math.abs(state.platenVy) * dt * 0.28);
                    } else {
                        state.temperatureC += (24 - state.temperatureC) * dt * 0.35;
                    }

                    const effectiveFractureStrain = spec.fractureStrain / dieMultiplier;
                    const failureLoadNorm = Math.min(1.0, state.totalStrain / effectiveFractureStrain);

                    // Hydraulic Hose Pulsation under High Pressure
                    const loadNorm = Math.min(1, state.forceTons / 900);
                    hoseMat.emissive.setHex(loadNorm > 0.7 ? 0xf43f5e : (loadNorm > 0.35 ? 0xf59e0b : 0x0284c7));
                    hoseMat.emissiveIntensity = 0.35 + loadNorm * 1.45;
                    if (loadNorm > 0.85) {
                        state.cameraShake = Math.max(state.cameraShake, 0.015 * loadNorm);
                    }

                    // Friction Sparks / Dust
                    if (inContact && Math.abs(state.platenVy) > 0.04 && failureLoadNorm > 0.4) {
                        if (Math.random() < 0.5) {
                            emitParticleBurst(ANVIL_TOP_Y + compressedHeight * 0.5, currentRadius, 3, 0.5, spec.sparkColor);
                        }
                    }

                    if (state.totalStrain >= effectiveFractureStrain) {
                        triggerSpecimenDestruction(spec, compressedHeight, currentRadius);
                    } else {
                        updateSpecimenGeometry(spec, state.totalStrain, failureLoadNorm);
                    }

                    if (inContact && Math.abs(state.platenVy) > 0.02) {
                        curvePoints.push({
                            strain: Math.min(1, state.totalStrain / 0.92),
                            force: Math.min(1, state.forceTons / 1000),
                            temp: Math.min(1, (state.temperatureC - 24) / 650)
                        });
                        if (curvePoints.length > 96) curvePoints.shift();
                    }

                    stressPointLight.intensity = 1.2 + failureLoadNorm * 3.8;
                    stressPointLight.color.setHex(spec.color);

                    if (failureLoadNorm > 0.65 && Math.abs(state.platenVy) > 0.05 && navigator.vibrate) {
                        const nowMs = performance.now();
                        if (nowMs - state.lastHapticTime > 120) {
                            navigator.vibrate(12);
                            state.lastHapticTime = nowMs;
                        }
                    }
                } else {
                    state.forceTons *= 0.82;
                    state.pressureMpa *= 0.82;
                    state.temperatureC += (24 - state.temperatureC) * dt * 0.5;
                    hoseMat.emissiveIntensity = 0.35;
                }

                // Step Contracts
                stepContracts(dt);

                // Update Hydraulic Audio
                const isMoving = Math.abs(state.platenVy) > 0.03;
                updateHydraulicAudio(isMoving, Math.min(1, state.forceTons / 800), state.timeScale);

                // Floor Shockwave Animation
                if (shockwaveProgress > 0) {
                    shockwaveProgress += dt * 2.4;
                    const s = 1.0 + shockwaveProgress * 9.5;
                    shockRingMesh.scale.set(s, s, 1);
                    shockRingMat.opacity = Math.max(0, 0.85 * (1 - shockwaveProgress));
                    if (shockwaveProgress >= 1.0) shockwaveProgress = 0;
                }

                // Flying 3D Shards Physics
                for (let i = 0; i < SHARD_COUNT; i++) {
                    const s = shards[i];
                    if (s.life > 0) {
                        s.life -= dt;
                        s.vy -= 9.8 * dt;
                        s.mesh.position.x += s.vx * dt;
                        s.mesh.position.y += s.vy * dt;
                        s.mesh.position.z += s.vz * dt;
                        s.mesh.rotation.x += s.rx * dt;
                        s.mesh.rotation.y += s.ry * dt;
                        s.mesh.rotation.z += s.rz * dt;

                        if (s.mesh.position.y < -1.78) {
                            s.mesh.position.y = -1.78;
                            s.vy = -s.vy * 0.35;
                            s.vx *= 0.72;
                            s.vz *= 0.72;
                        }
                        if (s.life <= 0) s.mesh.visible = false;
                    }
                }

                // Sparks / Dust Particles
                const pArr = partGeo.attributes.position.array;
                for (let i = 0; i < PARTICLE_COUNT; i++) {
                    if (partLife[i] > 0) {
                        partLife[i] -= dt * 1.9;
                        partVel[i].vy -= 4.5 * dt;
                        pArr[i * 3] += partVel[i].vx * dt;
                        pArr[i * 3 + 1] += partVel[i].vy * dt;
                        pArr[i * 3 + 2] += partVel[i].vz * dt;
                        if (partLife[i] <= 0) pArr[i * 3 + 1] = -99;
                    }
                }
                partGeo.attributes.position.needsUpdate = true;
            }

            // --- Touch & Mouse Interaction (Platen Drag, Specimen Laser Poke, 3D Orbit, Pinch Zoom) ---
            const raycaster = new THREE.Raycaster();
            const pointerNDC = new THREE.Vector2();
            const dragPlane = new THREE.Plane(new THREE.Vector3(0, 0, 1), 0);
            const planeHit = new THREE.Vector3();

            function updateNDC(clientX, clientY) {
                const rect = container.getBoundingClientRect();
                pointerNDC.x = ((clientX - rect.left) / rect.width) * 2 - 1;
                pointerNDC.y = -((clientY - rect.top) / rect.height) * 2 + 1;
            }

            function handlePointerStart(clientX, clientY) {
                updateNDC(clientX, clientY);
                raycaster.setFromCamera(pointerNDC, camera);

                const hitPlaten = raycaster.intersectObject(pressHitBox, false);
                if (hitPlaten.length > 0) {
                    state.dragMode = 'platen';
                    return;
                }

                // Direct Tap on Specimen -> Laser Hammer Impact!
                if (!state.destroyed) {
                    const hitSpec = raycaster.intersectObject(specimenMesh, false);
                    if (hitSpec.length > 0) {
                        const pt = hitSpec[0].point;
                        const spec = SPECIMENS[state.specIndex];
                        state.dentDepth = Math.min(0.18, state.dentDepth + 0.045);
                        state.temperatureC = Math.min(980, state.temperatureC + 28);
                        state.cameraShake = 0.07;
                        emitParticleBurst(pt.y, spec.restRadius, 22, 0.9, 0x38bdf8);
                        triggerLaserPulseSound();
                        updateSpecimenGeometry(spec, state.totalStrain, Math.min(1, state.totalStrain / spec.fractureStrain));
                        return;
                    }
                }

                state.dragMode = 'camera';
                state.lastPointerX = clientX;
                state.lastPointerY = clientY;
            }

            function handlePointerMove(clientX, clientY) {
                if (state.dragMode === 'platen') {
                    updateNDC(clientX, clientY);
                    raycaster.setFromCamera(pointerNDC, camera);
                    dragPlane.normal.set(Math.sin(camOrbit.theta), 0, Math.cos(camOrbit.theta)).normalize();
                    if (raycaster.ray.intersectPlane(dragPlane, planeHit)) {
                        const prevY = state.platenY;
                        const spec = SPECIMENS[state.specIndex];
                        const topLimit = Math.max(MAX_PLATEN_Y, ANVIL_TOP_Y + spec.restHeight + 0.36);
                        const targetY = Math.max(MIN_PLATEN_Y, Math.min(topLimit, planeHit.y - 0.15));
                        state.platenY += (targetY - state.platenY) * 0.35;
                        state.platenVy = (state.platenY - prevY) * 35;
                    }
                } else if (state.dragMode === 'camera') {
                    const dx = clientX - state.lastPointerX;
                    const dy = clientY - state.lastPointerY;
                    state.lastPointerX = clientX;
                    state.lastPointerY = clientY;

                    camOrbit.targetTheta -= dx * 0.0075;
                    camOrbit.targetPhi = Math.max(-0.35, Math.min(1.15, camOrbit.targetPhi + dy * 0.0075));
                }
            }

            function handlePointerEnd() {
                if (state.dragMode === 'platen') {
                    state.platenVy = 0;
                }
                state.dragMode = 'none';
            }

            container.addEventListener('mousedown', (e) => handlePointerStart(e.clientX, e.clientY));
            window.addEventListener('mousemove', (e) => handlePointerMove(e.clientX, e.clientY));
            window.addEventListener('mouseup', handlePointerEnd);

            container.addEventListener('wheel', (e) => {
                e.preventDefault();

                } else if (e.touches.length === 2) {
                    state.dragMode = 'pinch';
                    const dx = e.touches[0].clientX - e.touches[1].clientX;
                    const dy = e.touches[0].clientY - e.touches[1].clientY;
                    state.pinchStartDist = Math.hypot(dx, dy);
                    state.pinchStartZoom = camOrbit.targetRadius;
                }
            }, { passive: true });

            window.addEventListener('touchmove', (e) => {
                if (e.touches.length === 1 && (state.dragMode === 'platen' || state.dragMode === 'camera')) {
                    handlePointerMove(e.touches[0].clientX, e.touches[0].clientY);
                } else if (e.touches.length === 2 && state.dragMode === 'pinch') {
                    const dx = e.touches[0].clientX - e.touches[1].clientX;
                    const dy = e.touches[0].clientY - e.touches[1].clientY;
                    const dist = Math.hypot(dx, dy);
                    if (state.pinchStartDist > 10) {
                        camOrbit.targetRadius = Math.max(3.2, Math.min(11.0, state.pinchStartZoom * (state.pinchStartDist / dist)));
                    }
                }
            }, { passive: true });

                if (mode === 'orbit') {

                    camOrbit.targetPhi = 0.16;
                    camOrbit.targetRadius = baseDist;
                    camOrbit.targetLookY = -0.22;
                } else if (mode === 'macro') {
                    camOrbit.targetTheta = 0.34;
                    camOrbit.targetPhi = 0.12;
                    camOrbit.targetRadius = baseDist * 0.54;
                    camOrbit.targetLookY = -0.68;
                } else if (mode === 'side') {
                    camOrbit.targetTheta = 0.0;
                    camOrbit.targetPhi = 0.02;
                    camOrbit.targetRadius = baseDist * 0.94;
                    camOrbit.targetLookY = -0.34;
                } else if (mode === 'top') {
                    camOrbit.targetTheta = 0.02;
                    camOrbit.targetPhi = 1.08;
                    camOrbit.targetRadius = baseDist * 0.92;
                    camOrbit.targetLookY = -0.45;
                }
            }
            camButtons.forEach((b) => {
                b.addEventListener('click', () => setCameraPreset(b.getAttribute('data-cam')));
            });

            // --- Resize & 60fps Animation Loop ---
            function handleResize() {
                const width = container.clientWidth;
                const height = container.clientHeight;
                if (width === 0 || height === 0) return;
                camera.aspect = width / height;
                camera.updateProjectionMatrix();
                renderer.setSize(width, height);
            }
            window.addEventListener('resize', handleResize);

            let lastTimestamp = performance.now();

            function animate(now) {
                requestAnimationFrame(animate);

                const rawDt = Math.min(0.04, (now - lastTimestamp) / 1000);
                lastTimestamp = now;
                state.time += rawDt;

                stepPhysics(rawDt);
                updateUI();

                // Camera Orbit + Auto Slow-Mo Zoom + Impact Shake
                const slowZoomOffset = state.autoSlowMoTimer > 0 ? -1.1 : 0;
                camOrbit.theta += (camOrbit.targetTheta - camOrbit.theta) * 0.14;
                camOrbit.phi += (camOrbit.targetPhi - camOrbit.phi) * 0.14;
                camOrbit.radius += ((camOrbit.targetRadius + slowZoomOffset) - camOrbit.radius) * 0.14;
                camOrbit.lookY += (camOrbit.targetLookY - camOrbit.lookY) * 0.14;

                state.cameraShake *= Math.exp(-10 * rawDt);
                const shakeX = (Math.random() - 0.5) * state.cameraShake;
                const shakeY = (Math.random() - 0.5) * state.cameraShake;

                camera.position.x = camOrbit.radius * Math.sin(camOrbit.theta) * Math.cos(camOrbit.phi) + shakeX;
                camera.position.y = camOrbit.lookY + camOrbit.radius * Math.sin(camOrbit.phi) + shakeY;
                camera.position.z = camOrbit.radius * Math.cos(camOrbit.theta) * Math.cos(camOrbit.phi);
                camera.lookAt(0, camOrbit.lookY, 0);

                renderer.render(scene, camera);
            }

            window.onload = function () {
                handleResize();
                setDieType('flat');
                spawnSpecimen(0);
                setCameraPreset('orbit');
                camOrbit.theta = camOrbit.targetTheta;
                camOrbit.phi = camOrbit.targetPhi;
                camOrbit.radius = camOrbit.targetRadius;
                camOrbit.lookY = camOrbit.targetLookY;
                lastTimestamp = performance.now();
                requestAnimationFrame(animate);
            };
        })();
    </script>
